┌─────────────────────────────────────────────────────────────────────────┐
│ 🔧 API                                                                    │
└─────────────────────────────────────────────────────────────────────────┘

  # Получить текущую тему
  theme_manager.current -> Theme.DARK / Theme.LIGHT
  
  # Проверить темную ли тема
  theme_manager.is_dark -> bool
  
  # Получить токены
  theme_manager.tokens -> dict
  
  # Переключить тему
  theme_manager.toggle(app)
  
  # Установить конкретную тему
  theme_manager.set_theme(Theme.DARK, app)
  theme_manager.set_theme(Theme.LIGHT, app)
  
  # Подписаться на изменения
  theme_manager.theme_changed.connect(callback)

│ ✅ BEST PRACTICES                                                        │

  ✓ Всегда используйте токены, не хардкодьте цвета
  ✓ Используйте objectName для стилизации
  ✓ Слушайте theme_changed для обновления UI
  ✓ Группируйте элементы в Card
  ✓ Используйте правильные font-weight и spacing

  ✗ Не используйте хардкод цветов (#fff, #000)
  ✗ Не применяйте стили без токенов
  ✗ Не игнорируйте изменения темы
